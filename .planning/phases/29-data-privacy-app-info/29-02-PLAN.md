Create a simple credits screen matching the app's visual style (Color.appBg background, brand fonts). Include:

```swift
import SwiftUI

struct AboutView: View {
    private var appVersion: String {
        let version = Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "1.0"
        let build = Bundle.main.infoDictionary?["CFBundleVersion"] as? String ?? "1"
        return "Version \(version) (\(build))"
    }

    var body: some View {
        ZStack {
            Color.appBg.ignoresSafeArea()

            VStack(spacing: 24) {
                Spacer()

                Image("MascotNeutral")
                    .resizable()
                    .scaledToFit()
                    .frame(width: 120, height: 120)

                VStack(spacing: 8) {
                    Text("Abimo")
                        .font(.system(size: 32, weight: .bold, design: .rounded))
                        .foregroundColor(.textPri)

                    Text("Turn your ideas into action")
                        .font(.system(size: 16, weight: .medium))
                        .foregroundColor(.textSec)
                }

                Text(appVersion)
                    .font(.system(size: 14))
                    .foregroundColor(.textSec)

                Spacer()

                VStack(spacing: 4) {
                    Text("Made with love")
                        .font(.system(size: 14, weight: .medium))
                        .foregroundColor(.textSec)
                }
                .padding(.bottom, 32)
            }
        }
        .navigationTitle("About")
        .navigationBarTitleDisplayMode(.inline)
        .toolbarBackground(Color.appBg, for: .navigationBar)
        .toolbarColorScheme(.light, for: .navigationBar)
    }
}
```

Note: Uses "MascotNeutral" image asset which already exists in the project (used by MascotLoadingView and notification copy system).

```swift
// About section
settingsSection(title: "About") {
    settingsRow(icon: "info.circle", title: "App Version", color: .textSec, detail: appVersion)
    Divider().padding(.leading, 44)
    Button {
        requestReview()
    } label: {
        settingsRow(icon: "star", title: "Rate the App", color: .brandAmber)
    }
    Divider().padding(.leading, 44)
    Button {
        if let url = URL(string: "mailto:feedback@abimo.app?subject=Abimo%20Feedback") {
            UIApplication.shared.open(url)
        }
    } label: {
        settingsRow(icon: "envelope", title: "Send Feedback", color: .accentBlue)
    }
    Divider().padding(.leading, 44)
    NavigationLink(destination: AboutView()) {
        settingsRow(icon: "heart", title: "About", color: .brand)
    }
}
```

Notes:
- `requestReview()` uses the iOS 16+ environment action (app targets iOS 17+). Apple rate-limits this to 3 times per 365 days -- this is expected behavior, not a bug (per Pitfall 4 from research). INFO-02.
- `mailto:` URL uses placeholder email `feedback@abimo.app` with pre-filled subject. This pattern already exists in the codebase at ActionDetailSheet.swift line 201. INFO-03.
- INFO-01 is already implemented (appVersion computed property + display row). No changes needed.
- NavigationLink to AboutView works because SettingsView is already inside a NavigationStack (pushed from RootView). INFO-04.
